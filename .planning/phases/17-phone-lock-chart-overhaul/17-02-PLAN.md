**4. Add lock snapshot tracking and timestamped notification state** — inside the component body:
```tsx
const prevFocusRef = useRef<FocusArea>(focus);
const lockSnapshotRef = useRef<Record<string, number>>({});

// Each notification carries an id and arrival timestamp for auto-dismiss
interface LockNotif {
  id: string;
  contactName: string;
  preview: string;
  arrivedAt: number;
}
const [lockNotifs, setLockNotifs] = useState<LockNotif[]>([]);

useEffect(() => {
  if (prevFocusRef.current === 'phone' && focus !== 'phone') {
    // Phone just became locked — snapshot current message counts and clear old notifs
    const snapshot: Record<string, number> = {};
    Object.values(threads).forEach(t => {
      snapshot[t.contactName] = t.messages.length;
    });
    lockSnapshotRef.current = snapshot;
    setLockNotifs([]);
  }
  prevFocusRef.current = focus;
}, [focus, threads]);
```

**5. Detect new messages while locked and add timestamped notifications:**
```tsx
useEffect(() => {
  if (focus === 'phone') return; // phone is unlocked — no notifications needed

  const now = Date.now();
  const newNotifs: LockNotif[] = Object.values(threads)
    .filter(t => t.messages.length > (lockSnapshotRef.current[t.contactName] ?? 0))
    .map(t => ({
      id: `${t.contactName}-${t.messages.length}`,
      contactName: t.contactName,
      preview: t.messages[t.messages.length - 1]?.text ?? '',
      arrivedAt: now,
    }));

  if (newNotifs.length === 0) return;

  setLockNotifs(prev => {
    // Merge: keep existing (not yet expired) + add new ones, deduplicate by id, cap at 3
    const merged = [...prev, ...newNotifs.filter(n => !prev.some(p => p.id === n.id))];
    return merged.slice(-3);
  });

  // Update snapshot so same messages don't re-trigger
  newNotifs.forEach(n => {
    const thread = Object.values(threads).find(t => t.contactName === n.contactName);
    if (thread) {
      lockSnapshotRef.current = {
        ...lockSnapshotRef.current,
        [n.contactName]: thread.messages.length,
      };
    }
  });
}, [focus, threads]);
```

**6. Auto-dismiss notifications after 5 seconds** — after the above effects:
```tsx
useEffect(() => {
  if (lockNotifs.length === 0) return;

  const timer = setTimeout(() => {
    const cutoff = Date.now() - 5000;
    setLockNotifs(prev => prev.filter(n => n.arrivedAt > cutoff));
  }, 5100); // Check slightly after 5s to ensure all expired ones are caught

  return () => clearTimeout(timer);
}, [lockNotifs]);
```
